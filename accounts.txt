-- ================== CHỐNG CHẠY TRÙNG (theo server) ==================
-- Dùng JobId thay vì cờ true/false để teleport sang server mới vẫn chạy lại được
if getgenv().TCFG_JobId == game.JobId then return end
getgenv().TCFG_JobId = game.JobId

-- ================== THÔNG BÁO TỪNG BƯỚC ==================
local function notify(t)
    print("[Autoexec] " .. t)
    pcall(function()
        game:GetService("StarterGui"):SetCore("SendNotification", {
            Title = "Autoexec",
            Text = t,
            Duration = 4
        })
    end)
end
notify("Script bắt đầu chạy")

-- ================== ĐỢI GAME / NGƯỜI CHƠI ==================
if not game:IsLoaded() then
    game.Loaded:Wait()
end
notify("Game đã load")

local Players = game:GetService("Players")
local plr = Players.LocalPlayer
if not plr then
    Players:GetPropertyChangedSignal("LocalPlayer"):Wait()
    plr = Players.LocalPlayer
end

plr:WaitForChild("PlayerGui")

local waited = 0
while not plr.Character and waited < 10 do
    task.wait(0.25)
    waited += 0.25
end

task.wait(1) -- nếu script lỗi do load sớm thì tăng lên 2-3
notify("Người chơi sẵn sàng")

-- Tắt render 3D sớm để máy đỡ nặng
pcall(function()
    game:GetService("RunService"):Set3dRenderingEnabled(false)
end)

-- ================== LOADER ==================
local url = "https://api.luarmor.net/files/v4/loaders/5bfd68232034676923088ca8b6698be7.lua"
local success = false

notify("Đang tải loader...")
for attempt = 1, 5 do
    local ok, err = pcall(function()
        loadstring(game:HttpGet(url))()
    end)
    if ok then
        success = true
        break
    end
    warn("[Autoexec] Lần " .. attempt .. " lỗi: " .. tostring(err))
    notify("Lần " .. attempt .. " lỗi, thử lại...")
    task.wait(1)
end

if success then
    notify("Loader chạy xong")
else
    getgenv().TCFG_JobId = nil
    notify("Tải script thất bại sau 5 lần thử")
end
