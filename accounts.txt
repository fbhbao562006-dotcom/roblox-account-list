-- Chặn chạy 2 lần (đặt đầu tiên để không tốn thời gian chờ vô ích)
if getgenv().TCFG_Loaded then return end
getgenv().TCFG_Loaded = true

-- Đợi game load xong
if not game:IsLoaded() then
    game.Loaded:Wait()
end

-- Đợi người chơi + GUI + nhân vật (có giới hạn thời gian, không chờ cứng)
local Players = game:GetService("Players")
local plr = Players.LocalPlayer
if not plr then
    Players:GetPropertyChangedSignal("LocalPlayer"):Wait()
    plr = Players.LocalPlayer
end

plr:WaitForChild("PlayerGui")

local waited = 0
while not plr.Character and waited < 10 do
    task.wait(0.25)
    waited += 0.25
end

task.wait(1) -- nếu script bị lỗi do load sớm thì tăng lên 2-3

-- Tắt render 3D sớm để máy đỡ nặng lúc load
pcall(function()
    game:GetService("RunService"):Set3dRenderingEnabled(false)
end)

-- ================== LOADER ==================
local url = "https://api.luarmor.net/files/v4/loaders/5bfd68232034676923088ca8b6698be7.lua"
local success = false

for attempt = 1, 5 do
    local ok, err = pcall(function()
        loadstring(game:HttpGet(url))()
    end)
    if ok then
        success = true
        break
    end
    warn("[Autoexec] Lần " .. attempt .. " lỗi: " .. tostring(err))
    task.wait(1)
end

if not success then
    getgenv().TCFG_Loaded = nil
    warn("[Autoexec] Tải script thất bại sau 5 lần thử")
end
